cmake_minimum_required(VERSION 3.16)
project(titus_core VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/core/data_profiles.json")
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/core/data_profiles.json" TITUS_PROFILES)
configure_file(core/profiles.hpp.in generated/profiles.hpp @ONLY)
add_library(titus_core core/catalog.cpp core/sha256.cpp)
target_include_directories(titus_core PUBLIC include third_party/nlohmann-json PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
if(MSVC)
  target_compile_options(titus_core PRIVATE /W4 /utf-8)
else()
  target_compile_options(titus_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(titus-catalog core/main.cpp)
target_link_libraries(titus-catalog PRIVATE titus_core)
if(MINGW)
  target_link_options(titus-catalog PRIVATE -municode -static)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(check_catalog tests/check_catalog.cpp)
  target_link_libraries(check_catalog PRIVATE titus_core)
  if(MINGW)
    target_link_options(check_catalog PRIVATE -static)
  endif()
  add_test(NAME catalog COMMAND check_catalog "${CMAKE_CURRENT_SOURCE_DIR}/contracts/fingerprint-v1-vector.json")
endif()
install(TARGETS titus-catalog RUNTIME DESTINATION bin)
install(FILES third_party/nlohmann-json/LICENSE.MIT
        DESTINATION share/licenses/titus-engine/nlohmann-json)

add_library(titus_library launcher/library.cpp)
target_include_directories(titus_library PUBLIC launcher)
target_link_libraries(titus_library PUBLIC titus_core)
if(MSVC)
  target_compile_options(titus_library PRIVATE /utf-8)
endif()
option(TITUS_BUILD_LAUNCHER "Build the SDL2 global library frontend" OFF)
if(TITUS_BUILD_LAUNCHER)
  find_package(SDL2 2.30 CONFIG REQUIRED)
  find_path(TITUS_TTF_INCLUDE SDL_ttf.h PATH_SUFFIXES SDL2)
  find_library(TITUS_TTF_LIBRARY NAMES SDL2_ttf SDL2_ttf-2.0 libSDL2_ttf-2.0.so.0)
  if(NOT TITUS_TTF_INCLUDE OR NOT TITUS_TTF_LIBRARY)
    message(FATAL_ERROR "SDL2_ttf development files required; set TITUS_TTF_INCLUDE/TITUS_TTF_LIBRARY or install libsdl2-ttf-dev")
  endif()
  add_executable(titus-engine WIN32 launcher/app.cpp)
  target_include_directories(titus-engine PRIVATE "${TITUS_TTF_INCLUDE}")
  target_link_libraries(titus-engine PRIVATE titus_library SDL2::SDL2 "${TITUS_TTF_LIBRARY}")
  if(MINGW)
    target_link_options(titus-engine PRIVATE -municode -static)
  endif()
  if(MSVC)
    target_compile_options(titus-engine PRIVATE /utf-8)
  endif()
  if(WIN32)
    target_link_libraries(titus-engine PRIVATE shell32)
    set(TITUS_TTF_DLL "" CACHE FILEPATH "SDL2_ttf runtime DLL for the Windows frontend")
    add_custom_command(TARGET titus-engine POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
      "$<TARGET_FILE:SDL2::SDL2>" "$<TARGET_FILE_DIR:titus-engine>")
    if(TITUS_TTF_DLL)
      add_custom_command(TARGET titus-engine POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
        "${TITUS_TTF_DLL}" "$<TARGET_FILE_DIR:titus-engine>")
      install(FILES "${TITUS_TTF_DLL}" "$<TARGET_FILE:SDL2::SDL2>" DESTINATION bin)
    endif()
  endif()
  file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/assets/fonts")
  configure_file(assets/fonts/DejaVuSans.ttf assets/fonts/DejaVuSans.ttf COPYONLY)
  configure_file(assets/fonts/LICENSE-DejaVu.txt assets/fonts/LICENSE-DejaVu.txt COPYONLY)
  install(TARGETS titus-engine RUNTIME DESTINATION bin)
  install(DIRECTORY assets/fonts DESTINATION bin/assets)
endif()
if(BUILD_TESTING)
  add_executable(check_library tests/check_library.cpp)
  target_link_libraries(check_library PRIVATE titus_library)
  if(MINGW)
    target_link_options(check_library PRIVATE -static)
  endif()
  add_test(NAME library COMMAND check_library)
endif()
