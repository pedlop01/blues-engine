if(TITUS_BUILD_BLUES_RUNTIME OR TITUS_BUILD_FOX_RUNTIME OR TITUS_BUILD_PRE1_RUNTIME)

    string(REPLACE "g++" "gcc" TITUS_NATIVE_CC "${CMAKE_CXX_COMPILER}")
    if(EXISTS "${TITUS_NATIVE_CC}")
      set(CMAKE_C_COMPILER "${TITUS_NATIVE_CC}" CACHE FILEPATH "Native runtime C compiler")
    endif()
  endif()
  enable_language(C)
  if(MINGW AND CMAKE_C_COMPILER_ID STREQUAL "GNU" AND CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
    execute_process(COMMAND "${CMAKE_C_COMPILER}" -print-libgcc-file-name OUTPUT_VARIABLE TITUS_C_LIBGCC OUTPUT_STRIP_TRAILING_WHITESPACE)
    execute_process(COMMAND "${CMAKE_CXX_COMPILER}" -print-libgcc-file-name OUTPUT_VARIABLE TITUS_CXX_LIBGCC OUTPUT_STRIP_TRAILING_WHITESPACE)
    if(NOT TITUS_C_LIBGCC STREQUAL TITUS_CXX_LIBGCC)
      message(FATAL_ERROR "Use matching MinGW C/C++ runtimes (e.g. gcc-posix and g++-posix); mixing win32/posix changes libstdc++ stream layout")
    endif()
  endif()

if(TITUS_BUILD_BLUES_RUNTIME)
  find_package(SDL2 2.30 CONFIG REQUIRED)
  set(TITUS_MODPLUG_ROOT "" CACHE PATH "Optional libmodplug prefix (MinGW or native)")
  find_path(TITUS_MODPLUG_INCLUDE libmodplug/modplug.h HINTS "${TITUS_MODPLUG_ROOT}/include")
  find_library(TITUS_MODPLUG_LIBRARY NAMES modplug HINTS "${TITUS_MODPLUG_ROOT}/lib")
  if(NOT TITUS_MODPLUG_INCLUDE OR NOT TITUS_MODPLUG_LIBRARY)
    message(FATAL_ERROR "libmodplug development files required for Blues runtime")
  endif()
  add_library(blues_native STATIC engines/blues-family/bridge.c
    bb/decode.c bb/game.c bb/level.c bb/objects.c bb/state.c bb/resource.c bb/screen.c bb/sound.c bb/staticres.c bb/tiles.c bb/unpack.c
    mixer.c dos_audio.c third_party/nuked-opl3/opl3.c sys_sdl2.c util.c)
  target_include_directories(blues_native PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}" "${TITUS_MODPLUG_INCLUDE}")
  target_compile_definitions(blues_native PRIVATE TITUS_MANAGED BLUES_ONLY)
  target_compile_options(blues_native PRIVATE -funwind-tables)
  if(WIN32)
    target_compile_definitions(blues_native PRIVATE MODPLUG_STATIC)
  endif()
  target_link_libraries(blues_native PUBLIC SDL2::SDL2 "${TITUS_MODPLUG_LIBRARY}")
  find_package(Python3 COMPONENTS Interpreter REQUIRED)
  foreach(module IN ITEMS ja p2)
    file(GLOB module_sources CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/${module}/*.c")
    add_library(native_${module} OBJECT ${module_sources})
    target_include_directories(native_${module} PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
    target_compile_definitions(native_${module} PRIVATE TITUS_MANAGED)
    target_compile_options(native_${module} PRIVATE -funwind-tables)
    set(module_object "${CMAKE_CURRENT_BINARY_DIR}/game_${module}_managed.o")
    add_custom_command(OUTPUT "${module_object}"
      COMMAND "${CMAKE_LINKER}" -r -o "${module_object}" $<TARGET_OBJECTS:native_${module}>
      COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tools/namespace_coff_module.py"
        "${module_object}" "game_${module}" --nm "${CMAKE_NM}" --objcopy "${CMAKE_OBJCOPY}"
      DEPENDS native_${module} $<TARGET_OBJECTS:native_${module}> tools/namespace_coff_module.py
      COMMAND_EXPAND_LISTS VERBATIM)
    set_source_files_properties("${module_object}" PROPERTIES EXTERNAL_OBJECT TRUE GENERATED TRUE)
    list(APPEND TITUS_FAMILY_OBJECTS "${module_object}")
  endforeach()
  execute_process(COMMAND git rev-parse HEAD WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}" OUTPUT_VARIABLE TITUS_RUNTIME_REV OUTPUT_STRIP_TRAILING_WHITESPACE)
  add_executable(blues-family-runtime engines/blues-family/runtime.cpp ${TITUS_FAMILY_OBJECTS})
  target_link_libraries(blues-family-runtime PRIVATE titus_sessions blues_native)
  target_compile_definitions(blues-family-runtime PRIVATE TITUS_BUILD_REVISION="${TITUS_RUNTIME_REV}-th07")
  set_target_properties(blues-family-runtime PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/engines/blues-family")
  if(MINGW)
    target_link_options(blues-family-runtime PRIVATE -municode -static)
  endif()
  if(WIN32)
    add_custom_command(TARGET blues-family-runtime POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "$<TARGET_FILE:SDL2::SDL2>" "$<TARGET_FILE_DIR:blues-family-runtime>")
  endif()
  if(BUILD_TESTING)
    add_executable(check_mixer_headroom tests/check_mixer_headroom.c)
    target_include_directories(check_mixer_headroom PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}" "${TITUS_MODPLUG_INCLUDE}")
    target_link_libraries(check_mixer_headroom PRIVATE "${TITUS_MODPLUG_LIBRARY}")

      target_link_libraries(check_mixer_headroom PRIVATE stdc++ m)
    endif()
    add_test(NAME mixer_headroom COMMAND check_mixer_headroom)
  endif()
  # Registry declares capabilities per implemented game, not per shared engine.
  file(READ "${CMAKE_CURRENT_SOURCE_DIR}/engines/blues-family/registry.json.in" TITUS_FAMILY_TEMPLATE)
  string(CONFIGURE "${TITUS_FAMILY_TEMPLATE}" TITUS_FAMILY_REGISTRY_ENTRY @ONLY)
